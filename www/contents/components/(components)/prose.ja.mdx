## 使い方

:::code-group

```tsx [package]
import { Prose } from "@yamada-ui/react"
```

```tsx [alias]
import { Prose } from "@/components/ui"
```

```tsx [monorepo]
import { Prose } from "@workspaces/ui"
```

:::

```tsx
<Prose />
```

### バリアントを変更する

#### コード

```tsx preview
<VStack>
  <For each={["solid", "subtle", "surface", "outline"]}>
    {(code, index) => (
      <Prose key={index} code={code}>
        <p>
          <code>プレイヤー情報</code>を確認するには、
          <kbd>Enter</kbd>を押してください。
        </p>
      </Prose>
    )}
  </For>
</VStack>
```

```tsx preview
<Prose>
```

```tsx preview
  <div className="not-prose">
    <h2>ハンター協会の掲示板</h2>
```
