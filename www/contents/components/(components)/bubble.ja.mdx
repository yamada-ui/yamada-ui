```tsx preview
  <Bubble.Item direction="end">

    </Bubble.Reactions>

  <Bubble.Group>
    <Bubble.Item>…</Bubble.Item>

  </Bubble.Group>

```

## 使い方

:::code-group

```tsx [package]
import { Bubble } from "@yamada-ui/react"
```

```tsx [alias]
import { Bubble } from "@/components/ui"
```

```tsx [monorepo]
import { Bubble } from "@workspaces/ui"
```

:::

```tsx
<Bubble.Root>
  <Bubble.Group>
    <Bubble.Item>
      <Bubble.Content />
      <Bubble.Reactions />
    </Bubble.Item>
  </Bubble.Group>
</Bubble.Root>
```

:::note
`Bubble.Content`は省略することができます。
:::

### itemsを使う

```tsx preview functional
const items = useMemo<Bubble.Item[]>(
  () => [
    { children: "何者だ、お前は" },
    {
      children: "趣味でヒーローをやっている者だ",
      direction: "end",
      reactions: "👍",
      reactionsProps: { "aria-label": "Reactions: thumbs up", role: "img" },
    },
    { items: [{ children: "…" }, { children: "なんだ、その適当な設定は…" }] },
  ],
  [],
)

return <Bubble.Root items={items} endItemProps={{ variant: "solid" }} />
```

### バリアントを変更する

```tsx preview
<VStack>
  <For each={["solid", "subtle", "surface", "outline", "panel", "ghost"]}>
    {(variant, index) => (
      <Bubble.Root key={index} variant={variant}>
        <Bubble.Item>ヒーローが逃げたら誰が戦うんだよ</Bubble.Item>
      </Bubble.Root>
    )}
  </For>
</VStack>
```

### サイズを変更する

```tsx preview
<VStack>
  <For each={["sm", "md", "lg"]}>
    {(size, index) => (
      <Bubble.Root key={index} size={size} variant="solid" py="md">
        <Bubble.Item>
          <Bubble.Content>
            退屈な人生に刺激がほしくて他の星を襲うなんてOLでも考えねえぞ
          </Bubble.Content>
          <Bubble.Reactions
            aria-label="Reactions: thumbs up, fire, and 2 more"
            role="img"
          >
            <span>👍</span>
            <span>🔥</span>
            <span>+2</span>
          </Bubble.Reactions>
        </Bubble.Item>
      </Bubble.Root>
    )}
  </For>
</VStack>
```

### カラースキームを変更する

```tsx preview
<VStack>
  <For each={["success", "warning"]}>
    {(colorScheme, index) => (
      <Bubble.Root key={index} colorScheme={colorScheme} variant="solid">
        <Bubble.Item>地球がヤバい!!!</Bubble.Item>
      </Bubble.Root>
    )}
  </For>
</VStack>
```

### グループ化する

グループ化する場合は、`Bubble.Group`を使用します。

```tsx preview
<Bubble.Root endItemProps={{ variant: "solid" }}>
  <Bubble.Item>ここに住んでもいいですか？</Bubble.Item>

  <Bubble.Group direction="end">
    <Bubble.Item>うん</Bubble.Item>
    <Bubble.Item>絶対ダメ</Bubble.Item>
  </Bubble.Group>

  <Bubble.Group>
    <Bubble.Item>ドサッ</Bubble.Item>
    <Bubble.Item>部屋代払います</Bubble.Item>
  </Bubble.Group>

  <Bubble.Item direction="end">ちゃんと歯ブラシ持ってきたか？</Bubble.Item>
</Bubble.Root>
```

### リアクションを表示する

リアクションを表示する場合は、`Bubble.Reactions`を使用します。配置を変更する場合は、`placement`を設定します。デフォルトは、`"end-end"`が設定されています。

```tsx preview
<VStack>
  <For
    each={[
      "start-start",
      "start-center",
      "start-end",
      "end-start",
      "end-center",
      "end-end",
    ]}
  >
    {(placement, index) => (
      <Bubble.Root key={index} variant="solid" py="md">
        <Bubble.Item>
          <Bubble.Content>
            ここで俺はお前に立ち向かわなくちゃいけないんだ！
          </Bubble.Content>
          <Bubble.Reactions
            aria-label="Reactions: thumbs up, fire, and 2 more"
            placement={placement}
            role="img"
          >
            <span>👍</span>
            <span>🔥</span>
            <span>+2</span>
          </Bubble.Reactions>
        </Bubble.Item>
      </Bubble.Root>
    )}
  </For>
</VStack>
```

## アクセシビリティ

操作を伴わない絵文字のリアクションを表示する場合は、`Bubble.Reactions`に`role="img"`とリアクションの内容を説明する`aria-label`を設定します。

```tsx
<Bubble.Root>
  <Bubble.Item>
    <Bubble.Content>
      ここで俺はお前に立ち向かわなくちゃいけないんだ！
    </Bubble.Content>
    <Bubble.Reactions
      aria-label="Reactions: thumbs up, fire, and 2 more"
      role="img"
    >
      <span>👍</span>
      <span>🔥</span>
      <span>+2</span>
    </Bubble.Reactions>
  </Bubble.Item>
</Bubble.Root>
```
