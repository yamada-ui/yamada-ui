## Usage

:::code-group

```tsx [package]
import { Marquee } from "@yamada-ui/react"
```

```tsx [alias]
import { Marquee } from "@/components/ui"
```

```tsx [monorepo]
import { Marquee } from "@workspaces/ui"
```

:::

```tsx
<Marquee.Root>
  <Marquee.Edge direction="start" />
  <Marquee.Content>
    <Marquee.Item />
  </Marquee.Content>
  <Marquee.Edge direction="end" />
</Marquee.Root>
```

:::note
`Marquee.Content` can be omitted.
:::

### Change Direction

To change the direction, set `orientation` to `"horizontal"` or `"vertical"`. By default, `"horizontal"` is set.

```tsx preview
<VStack>
  <For each={["horizontal", "vertical"]}>
    {(orientation, key) => (
      <Marquee.Root
        key={key}
        aria-label={`${orientation} brand logos`}
        contain="inline-size"
        gap="xl"
        h={orientation === "vertical" ? "xl" : undefined}
        orientation={orientation}
      >
        {marqueeItems.map(({ color, icon, label }) => (
          <Marquee.Item
            key={label}
            px={orientation === "vertical" ? "sm" : "xl"}
            py={orientation === "vertical" ? "xl" : "sm"}
          >
            <Icon
              as={icon}
              aria-hidden={false}
              aria-label={label}
              color={color}
              fontSize="6xl"
            />
          </Marquee.Item>
        ))}
      </Marquee.Root>
    )}
  </For>
</VStack>
```

### Change Scroll Direction

To change the scrolling direction, set `direction` to `"start"` or `"end"`. By default, `"start"` is set.

```tsx preview

              fontSize="6xl"
            />
          </Marquee.Item>

</VStack>
```

### Change Speed

To change the scrolling speed, set a numerical value (pixels per second) to `speed`. By default, `50` is set.

```tsx preview

      <Marquee.Root
        key={key}
        aria-label={`Brand logos at ${speed} pixels per second`}
        contain="inline-size"
        gap="xl"
        speed={speed}
      >
        {marqueeItems.map(({ color, icon, label }) => (
          <Marquee.Item key={label} px="xl" py="sm">
            <Icon
              as={icon}
              aria-hidden={false}
              aria-label={label}
              color={color}
              fontSize="6xl"
            />
          </Marquee.Item>
        ))}
      </Marquee.Root>
```

### Control

```tsx preview functional client
const [paused, setPaused] = useState(false)

return (
  <VStack alignItems="start">
    <Marquee.Root aria-label="Controlled brand logos" gap="xl" paused={paused}>
      {marqueeItems.map(({ color, icon, label }) => (
        <Marquee.Item key={label} px="xl" py="sm">
          <Icon
            as={icon}
            aria-hidden={false}
            aria-label={label}
            color={color}
            fontSize="6xl"
          />
        </Marquee.Item>
      ))}
    </Marquee.Root>

    <Button
      startIcon={paused ? <PlayIcon /> : <PauseIcon />}
      onClick={() => setPaused((prev) => !prev)}
    >
      {paused ? "Resume" : "Pause"}
    </Button>
  </VStack>
)
```

```tsx preview functional client
const controlRef = useRef<Marquee.Control>(null)

return (

    <Wrap gap="md">

    </Wrap>
  </VStack>
)
```

## Accessibility

When `aria-label` or `aria-labelledby` is set, it will be read aloud by screen readers.

```tsx
<Marquee.Root aria-label="Technologies">
  <Marquee.Item>Yamada UI</Marquee.Item>
  <Marquee.Item>React</Marquee.Item>
  <Marquee.Item>TypeScript</Marquee.Item>
</Marquee.Root>
```

For interactive content, set `pauseOnFocus` to `true` so keyboard users can access each item.

### Keyboard Navigation

| Key     | Description                                          | State |
| ------- | ---------------------------------------------------- | ----- |
| `Space` | Toggles pause and resume when the element has focus. | -     |

### ARIA Roles and Attributes

| Element                             | Roles and Attributes             | Usage                                                 |
| ----------------------------------- | -------------------------------- | ----------------------------------------------------- |
| `div.ui-marquee__root`              | `role="region"`                  | Indicates that this is a region.                      |
|                                     | `aria-roledescription="marquee"` | Indicates that this is a marquee.                     |
|                                     | `aria-live="off"`                | Prevents repeated announcements of scrolling content. |
|                                     | `aria-keyshortcuts="Space"`      | Indicates the keyboard shortcut to toggle pause.      |
| `div.ui-marquee__group[data-clone]` | `aria-hidden="true"`             | Hides duplicated content from screen readers.         |
|                                     | `inert`                          | Excludes duplicated content from keyboard navigation. |
| `div.ui-marquee__edge`              | `aria-hidden="true"`             | Hides decorative gradients from screen readers.       |
